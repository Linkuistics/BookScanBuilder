# Compare every pre-existing test/function/method, not an anticipated pattern list.
def segments(raw):
 tree=ast.parse(raw)
 lines=raw.splitlines(keepends=True)
 result={}
 for node in tree.body:
  if isinstance(node,ast.ClassDef):
   for method in node.body:
    if isinstance(method,(ast.FunctionDef,ast.AsyncFunctionDef)):
     begin=min([method.lineno]+[d.lineno for d in method.decorator_list])-1
     result[node.name+'.'+method.name]=''.join(lines[begin:method.end_lineno])
  elif isinstance(node,(ast.FunctionDef,ast.AsyncFunctionDef,ast.Import,ast.ImportFrom)):
   begin=min([node.lineno]+[d.lineno for d in getattr(node,'decorator_list',[])])-1
   key=getattr(node,'name',ast.dump(node,include_attributes=False))
   result[key]=''.join(lines[begin:node.end_lineno])
 return result
