import hashlib,json,os,pathlib,shutil,subprocess,xml.etree.ElementTree as ET
root=pathlib.Path.cwd()
out=pathlib.Path('/private/tmp/bookscan-k63-lock-cleanup/mutations-v2')
source=root/'src/bookscan/ownership.py'
raw=source.read_text()
mutations={
 'omit-cleanup':raw.replace('_remove_failed_lock(path, directory, lock, parent, captured)','pass'),
 'omit-root-checks':raw.replace('    verify_root()\n','    pass\n'),
 'float-time-identity':raw.replace('entry.st_mtime_ns','entry.st_mtime').replace('entry.st_ctime_ns','entry.st_ctime'),
 'trust-name':raw.replace('_stamp(named) != _stamp(captured)','False').replace('if _stamp(os.stat(LOCK_NAME, dir_fd=directory, follow_symlinks=False)) != _stamp(captured):','if False:'),
 'replace-selected-on-close':raw.replace('selected = sys.exception()','selected = None'),
}
results=[]
for name,changed in mutations.items():
 assert changed!=raw,name
 directory=out/name
 package=directory/'package'
 package.mkdir(parents=True)
 shutil.copytree(root/'src/bookscan',package/'bookscan',ignore=shutil.ignore_patterns('__pycache__'))
 (package/'bookscan/ownership.py').write_text(changed)
 xml=directory/'cases.xml'
 command=[str(root/'.venv/bin/python'),'-m','pytest','-q',str(root/'tests/test_lock_cleanup.py'),str(root/'tests/test_ownership.py'),'--junitxml='+str(xml)]
 env=os.environ.copy();env['PYTHONPATH']=str(package);env.pop('PYTEST_ADDOPTS',None)
 with (directory/'output.log').open('wb') as log:
  result=subprocess.run(command,env=env,stdout=log,stderr=subprocess.STDOUT)
 cases=[(c.attrib['classname'],c.attrib['name'],'failed' if c.find('failure') is not None else 'error' if c.find('error') is not None else 'skipped' if c.find('skipped') is not None else 'passed') for c in ET.parse(xml).iter('testcase')]
 failed=[c for c in cases if c[2]=='failed']
 assert result.returncode==1 and failed and all(c[2] in {'failed','passed'} for c in cases),name
 assert all(c[0].rsplit('.',1)[-1]=='test_lock_cleanup' for c in failed),failed
 results.append({'name':name,'source_sha256':hashlib.sha256(changed.encode()).hexdigest(),'source':changed,'command':command,'exit':result.returncode,'cases':cases})
 print(name,len(failed),'failures',sum(c[2]=='passed' for c in cases),'passed',flush=True)
assert source.read_text()==raw
(out/'results.json').write_text(json.dumps(results,sort_keys=True,indent=2)+'\n')
